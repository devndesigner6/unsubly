# smart_contracts/escrow/contract.py
from algopy import ARC4Contract, GlobalState, Account, UInt64, arc4

class EscrowVault(ARC4Contract):
    creator: GlobalState[Account]
    recipient: GlobalState[Account]
    status: GlobalState[UInt64]  # 0=active, 1=released, 2=killed

    @arc4.abimethod(create="require")
    def create(self, recipient: Account) -> None:
        self.creator.value = Txn.sender
        self.recipient.value = recipient
        self.status.value = UInt64(0)

    @arc4.abimethod
    def release(self) -> None:
        assert Txn.sender == self.creator.value
        assert self.status.value == UInt64(0)
        # Inner txn to pay recipient
        ...
        self.status.value = UInt64(1)

    @arc4.abimethod
    def kill(self) -> None:
        assert Txn.sender == self.creator.value
        assert self.status.value == UInt64(0)
        # Inner txn to refund creator
        ...
        self.status.value = UInt64(2)

    @arc4.abimethod(allow_actions=["DeleteApplication"])
    def delete(self) -> None:
        assert Txn.sender == self.creator.value
        assert self.status.value != UInt64(0)
